Example

A route is a value

Name each screen with a small type, and let the coordinator decide where it goes.

ProductRoute.swift
// Conformances:
// - Route, so a coordinator can push it and logs can name it.
struct ProductRoute: Route {
    let id: Product.ID
}

func showProduct(id: Product.ID) {
    navigator.push(
        ProductRoute(id: id)
    )
}
